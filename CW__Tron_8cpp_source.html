84static void keccak_f(uint64_t* a) {
85 for (unsigned round = 0U; round < KECCAK_ROUNDS; round++) {
86 uint64_t c[5];
87 uint64_t d[5];
88 uint64_t b[KECCAK_STATE_LANES];
89
90 /* Theta */
91 for (unsigned x = 0U; x < 5U; x++) {
92 c[x] = a[x] ^ a[x + 5U] ^ a[x + 10U] ^ a[x + 15U] ^ a[x + 20U];
93 }
94 for (unsigned x = 0U; x < 5U; x++) {
95 d[x] = c[(x + 4U) % 5U] ^ ROTL64(c[(x + 1U) % 5U], 1U);
96 }
97 for (unsigned y = 0U; y < 5U; y++) {
98 for (unsigned x = 0U; x < 5U; x++) {
99 a[x + (5U * y)] ^= d[x];
100 }
101 }
102
103 /* Rho + Pi */
104 for (unsigned y = 0U; y < 5U; y++) {
105 for (unsigned x = 0U; x < 5U; x++) {
106 const unsigned src = x + (5U * y);
107 const unsigned dst = y + (5U * (((2U * x) + (3U * y)) % 5U));
108 b[dst] = (KECCAK_ROT[src] == 0U) ? a[src]
109 : ROTL64(a[src], KECCAK_ROT[src]);
110 }
111 }
112
113 /* Chi */
114 for (unsigned y = 0U; y < 5U; y++) {
115 for (unsigned x = 0U; x < 5U; x++) {
116 a[x + (5U * y)] = b[x + (5U * y)] ^
117 ((~b[((x + 1U) % 5U) + (5U * y)]) &
118 b[((x + 2U) % 5U) + (5U * y)]);
119 }
120 }
121
122 /* Iota */
123 a[0] ^= KECCAK_RC[round];
124 }
125}

133static void keccak_absorb(uint64_t* a, const uint8_t* data, size_t len) {
134 size_t offset = 0U;
135
136 while (len >= KECCAK_RATE_BYTES) {
137 for (size_t i = 0U; i < KECCAK_RATE_BYTES; i++) {
138 a[i / 8U] ^= ((uint64_t)data[offset + i]) << (8U * (i % 8U));
139 }
140 keccak_f(a);
141 offset += KECCAK_RATE_BYTES;
142 len -= KECCAK_RATE_BYTES;
143 }
144
145 /* Final (possibly empty) block plus Keccak padding: 0x01 ... 0x80. */
146 for (size_t i = 0U; i < len; i++) {
147 a[i / 8U] ^= ((uint64_t)data[offset + i]) << (8U * (i % 8U));
148 }
149 a[len / 8U] ^= ((uint64_t)0x01U) << (8U * (len % 8U));
150 a[(KECCAK_RATE_BYTES - 1U) / 8U] ^=
151 ((uint64_t)0x80U) << (8U * ((KECCAK_RATE_BYTES - 1U) % 8U));
152 keccak_f(a);
153}

155bool CW_Tron::keccak256(const uint8_t* data, size_t len, uint8_t* out) {
156 bool ret = false;
157
158 if (((data != NULL) || (len == 0U)) && (out != NULL)) {
159 uint64_t a[KECCAK_STATE_LANES];
160 for (unsigned i = 0U; i < KECCAK_STATE_LANES; i++) {
161 a[i] = 0ULL;
162 }
163
164 keccak_absorb(a, data, len);
165
166 for (unsigned i = 0U; i < KECCAK_DIGEST_BYTES; i++) {
167 out[i] = (uint8_t)((a[i / 8U] >> (8U * (i % 8U))) & 0xFFU);
168 }
169 ret = true;
170 }
171
172 return ret;
173}

179/*
180 * ponytail: no leading-zero ('1' padding) handling — a TRON payload always
181 * starts with 0x41, so the encoding is always exactly 34 chars. Add the
182 * generic leading-zero loop only if this is ever reused for Bitcoin-style
183 * payloads, which can start with 0x00.
184 */
185
192static bool base58_encode_payload(const uint8_t* payload, char* out) {
193 uint8_t tmp[TRON_PAYLOAD_BYTES];
194 char reversed[CW_TRON_ADDRESS_STR_LEN];
195 size_t digits = 0U;
196 size_t start = 0U;
197
198 memcpy(tmp, payload, sizeof(tmp));
199
200 while ((start < TRON_PAYLOAD_BYTES) && (digits < sizeof(reversed))) {
201 uint32_t remainder = 0U;
202 for (size_t i = start; i < TRON_PAYLOAD_BYTES; i++) {
203 const uint32_t acc = (remainder << 8U) | (uint32_t)tmp[i];
204 tmp[i] = (uint8_t)(acc / BASE58_BASE);
205 remainder = acc % BASE58_BASE;
206 }
207 reversed[digits] = BASE58_ALPHABET[remainder];
208 digits++;
209
210 while ((start < TRON_PAYLOAD_BYTES) && (tmp[start] == 0U)) {
211 start++;
212 }
213 }
214
215 const bool ret = (digits == CW_TRON_ADDRESS_STR_LEN);
216 if (ret) {
217 for (size_t i = 0U; i < digits; i++) {
218 out[i] = reversed[digits - 1U - i];
219 }
220 out[digits] = '\0';
221 }
222
223 return ret;
224}

232static bool base58_decode_payload(const char* address, uint8_t* payload) {
233 memset(payload, 0, TRON_PAYLOAD_BYTES);
234
235 for (size_t i = 0U; i < CW_TRON_ADDRESS_STR_LEN; i++) {
236 const char* found = strchr(BASE58_ALPHABET, address[i]);
237 /* strchr would also match the terminating NUL of the alphabet. */
238 if ((found == NULL) || (address[i] == '\0')) {
239 return false;
240 }
241
242 uint32_t carry = (uint32_t)(found - BASE58_ALPHABET);
243 for (size_t j = TRON_PAYLOAD_BYTES; j > 0U; j--) {
244 const uint32_t acc = ((uint32_t)payload[j - 1U] * BASE58_BASE) + carry;
245 payload[j - 1U] = (uint8_t)(acc & 0xFFU);
246 carry = acc >> 8U;
247 }
248 if (carry != 0U) {
249 return false; /* value wider than 25 bytes */
250 }
251 }
252
253 return true;
254}

263static bool tron_checksum(const uint8_t* addressBytes, CW_CryptoProvider& crypto,
264 uint8_t* checksum4) {
265 uint8_t first[KECCAK_DIGEST_BYTES];
266 uint8_t second[KECCAK_DIGEST_BYTES];
267 bool ret = false;
268
269 if (crypto.sha256(addressBytes, CW_TRON_ADDRESS_BYTES, first) &&
270 crypto.sha256(first, sizeof(first), second)) {
271 memcpy(checksum4, second, TRON_CHECKSUM_BYTES);
272 ret = true;
273 }
274
275 return ret;
276}

296 bool ret = false;
297
298 if ((addressBytes != NULL) && (out != NULL) &&

300 (addressBytes[0] == CW_TRON_ADDRESS_PREFIX)) {
301
302 uint8_t payload[TRON_PAYLOAD_BYTES];
303 memcpy(payload, addressBytes, CW_TRON_ADDRESS_BYTES);

322bool CW_Tron::decodeAddress(const char* address, CW_CryptoProvider& crypto,
323 uint8_t* out) {
324 bool ret = false;
325
326 if ((address != NULL) && (out != NULL) &&
327 (strlen(address) == CW_TRON_ADDRESS_STR_LEN)) {
328
329 uint8_t payload[TRON_PAYLOAD_BYTES];
330 uint8_t checksum[TRON_CHECKSUM_BYTES];
331
332 if (base58_decode_payload(address, payload) &&
333 (payload[0] == CW_TRON_ADDRESS_PREFIX) &&
334 tron_checksum(payload, crypto, checksum) &&
335 CW_Utils::secure_compare(checksum, &payload[CW_TRON_ADDRESS_BYTES],
336 TRON_CHECKSUM_BYTES)) {
337 memcpy(out, payload, CW_TRON_ADDRESS_BYTES);
338 ret = true;
339 }
340 }
341
342 return ret;
