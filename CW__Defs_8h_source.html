123/* Ed25519 signs the raw message (the card hashes internally). Cap the message
124 * so the framed payload [len(2)|msg|path|pin(9)] fits one secure-channel page.
125 *
126 * These two are WORST-CASE bounds, kept for reference and for callers that want a
127 * compile-time constant. CW_MAX_ED25519_MESSAGE_LENGTH reserves the full 20-byte
128 * path, so it under-reports the room available to a shorter path — a 4-level
129 * Solana path (16 B) leaves 181, not 177. validateSignRequest() therefore
130 * computes the bound from the actual request instead of using these; sizing a
131 * caller-side buffer with CW_MAX_ED25519_MESSAGE_LENGTH stays safe, just
132 * conservative.
133 *
134 * Neither figure is a card limit: the v2.0 SIGN spec accepts up to 1200 bytes of
135 * raw data. The ceiling is this SDK's single-page secure-channel buffer. */
136#define CW_MAX_ED25519_MESSAGE_LENGTH (CW_USER_DATA_PAGE_SIZE - 2U - CW_MAX_DERIVE_PATH_LENGTH - CW_MAX_PIN_LENGTH) /* 177 */
137#define CW_MAX_ED25519_MESSAGE_LENGTH_CURR (CW_USER_DATA_PAGE_SIZE - 2U - CW_MAX_PIN_LENGTH) /* 197 */
138#define CW_MAX_DERIVE_PATH_LENGTH (20U)
139#define CW_MIN_PIN_LENGTH (4U)
140#define CW_MAX_PIN_LENGTH (9U)
141#define CW_USER_DATA_PAGE_SIZE (208U)
142/* Largest plaintext one secure-channel exchange can hand back: a full page plus
143 * the block the bit padding may add. Bounds every decryptedOutput buffer. */
144#define CW_SECURE_PAYLOAD_MAX (CW_USER_DATA_PAGE_SIZE + 16U)
145#define CW_CONNECT_MAX_ATTEMPTS (5U)

200struct CW_SecureSession {
201 uint8_t aesKey[CW_AESKEY_SIZE] {};
202 uint8_t macKey[CW_MACKEY_SIZE] {};
203 uint8_t iv[CW_IV_SIZE] {};
204 uint32_t macCounter {};
205
207 CW_SecureSession() = default;
208
210 void clear() {
211 CW_Utils::secure_wipe(aesKey, sizeof(aesKey));
212 CW_Utils::secure_wipe(macKey, sizeof(macKey));
213 CW_Utils::secure_wipe(iv, sizeof(iv));
214 macCounter = 0U;
215 }
216};
