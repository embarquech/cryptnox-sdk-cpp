28class FakeNfc : public CW_NfcTransport {
29public:
30 uint8_t appletMajor = 2U; /* version byte returned by SELECT */
31 bool sendOk = false; /* result for every non-SELECT APDU */
32 uint8_t resp[255] = { 0U }; /* response for every non-SELECT APDU */
33 uint8_t respLen = 0U;
34 unsigned sent = 0U; /* non-SELECT APDUs sent */
35 uint8_t last[255] = { 0U }; /* last non-SELECT APDU */
36
37 bool begin() override { return true; }
38 bool inListPassiveTarget() override { return true; }
39 void resetReader() override { }
40 bool printFirmwareVersion() override { return true; }
41
42 bool sendAPDU(const uint8_t* apdu, uint8_t apduLen,
43 uint8_t* response, uint8_t& responseLen) override {
44 if ((apdu[0] == 0x00U) && (apdu[1] == 0xA4U)) {
45 const uint8_t sel[] = { 0x42U, appletMajor, 0x00U, 0x00U, 0x00U, 0x50U, 0x90U, 0x00U };
46 memcpy(response, sel, sizeof(sel));
47 responseLen = sizeof(sel);
48 return true;
49 }
50 sent++;
51 memcpy(last, apdu, apduLen);
52 memcpy(response, resp, respLen);
53 responseLen = respLen;
54 return sendOk;
55 }
56};

58class FakeCrypto : public CW_CryptoProvider {
59public:
60 bool failMacIv = false; /* fail the 16-byte counter block encryption (v2 MAC IV) */
61 bool failMac = false; /* fail the MAC computation itself */
62
63 bool sha256(const uint8_t*, size_t, uint8_t* out) override { memset(out, 0, 32); return true; }
64 bool sha512(const uint8_t*, size_t, uint8_t* out) override { memset(out, 0, 64); return true; }
65 bool ecdh(const uint8_t*, const uint8_t*, uint8_t* s, CW_Curve) override { memset(s, 1, 32); return true; }
66 bool makeKey(uint8_t* pub, uint8_t* priv, CW_Curve) override { memset(pub, 2, 64); memset(priv, 3, 32); return true; }
67 bool random(uint8_t* d, unsigned n) override { memset(d, 0x5A, n); return true; }
68 bool ecdsaVerify(const uint8_t*, const uint8_t*, size_t, const uint8_t*, CW_Curve) override { return false; }
69
70 uint16_t aesCbcEncrypt(const uint8_t* in, uint16_t len, uint8_t* out,
71 const uint8_t* key, uint8_t, uint8_t* iv, bool bitPadding) override {
72 if (!bitPadding && (len == 16U) && failMacIv) { return 0U; }
73 if (!bitPadding && (len > 16U) && failMac) { return 0U; }
74 uint8_t buf[512] = { 0U };
75 memcpy(buf, in, len);
76 uint16_t total = len;
77 if (bitPadding) {
78 buf[total++] = 0x80U;
79 while ((total % 16U) != 0U) { buf[total++] = 0U; }
80 }
81 for (uint16_t b = 0U; b < total; b += 16U) {
82 for (uint8_t i = 0U; i < 16U; i++) { iv[i] = (uint8_t)(buf[b + i] ^ iv[i] ^ key[i]); }
83 memcpy(out + b, iv, 16U);
84 }
85 return total;
86 }
87
88 uint16_t aesCbcDecrypt(uint8_t* in, uint16_t len, uint8_t* out,
89 const uint8_t* key, uint8_t, uint8_t* iv, bool bitPadding) override {
90 for (uint16_t b = 0U; b < len; b += 16U) {
91 for (uint8_t i = 0U; i < 16U; i++) {
92 out[b + i] = (uint8_t)(in[b + i] ^ key[i] ^ iv[i]);
93 iv[i] = in[b + i];
94 }
95 }
96 uint16_t n = len;
97 if (bitPadding) {
98 while ((n > 0U) && (out[n - 1U] == 0U)) { n--; }
99 if ((n > 0U) && (out[n - 1U] == 0x80U)) { n--; }
100 }
101 return n;
102 }

133static void openSession(CW_SecureSession& s) {
134 s.clear();
135 memset(s.aesKey, 0x11, sizeof(s.aesKey));
136 memset(s.macKey, 0x22, sizeof(s.macKey));
137 memset(s.iv, 0x33, sizeof(s.iv));
138}

140static bool isCleared(const CW_SecureSession& s) {
141 uint8_t acc = 0U;
142 for (uint8_t i = 0U; i < sizeof(s.aesKey); i++) { acc |= s.aesKey[i]; }
143 for (uint8_t i = 0U; i < sizeof(s.macKey); i++) { acc |= s.macKey[i]; }
144 return (acc == 0U) && (s.macCounter == 0U);
145}

150/******************************************************************

154/* Fresh fakes + a v2 channel + an open session for each case. */
155struct Rig {
156 FakeNfc nfc;
157 FakeCrypto crypto;
158 CW_SecureChannel ch;
159 CW_SecureSession s;
160 Rig() : ch(nfc, logger, crypto, platform) {
161 CHECK(ch.selectApdu() && (ch.appletMajor() == 2U));
162 openSession(s);
163 }
164 bool send(const uint8_t* data, uint16_t len) {
165 return ch.aesCbcEncrypt(s, HDR, sizeof(HDR), data, len);
166 }
167};

169static void check_session_clearing(void) {
170 { /* Exhausted counter: cleared, nothing sent. */
171 Rig r;
172 r.s.macCounter = UINT32_MAX;
173 CHECK(!r.send(DATA, sizeof(DATA)));
174 CHECK(isCleared(r.s) && (r.nfc.sent == 0U));
175 }
176 { /* MAC IV derivation failure: cleared, nothing sent. */
177 Rig r;
178 r.crypto.failMacIv = true;
179 CHECK(!r.send(DATA, sizeof(DATA)));
180 CHECK(isCleared(r.s) && (r.nfc.sent == 0U));
181 }
182 { /* MAC computation failure: cleared, nothing sent. */
183 Rig r;
184 r.crypto.failMac = true;
185 CHECK(!r.send(DATA, sizeof(DATA)));
186 CHECK(isCleared(r.s) && (r.nfc.sent == 0U));
187 }
188 { /* Transport failure: cleared. */
189 Rig r;
190 CHECK(!r.send(DATA, sizeof(DATA)));
191 CHECK(isCleared(r.s) && (r.nfc.sent == 1U));
192 }
193 { /* Bad response MAC (SW 9000, junk MAC): cleared. */
194 Rig r;
195 r.nfc.sendOk = true;
196 memset(r.nfc.resp, 0xEEU, 32U);
197 r.nfc.resp[32] = 0x90U;
198 r.nfc.resp[33] = 0x00U;
199 r.nfc.respLen = 34U;
200 CHECK(!r.send(DATA, sizeof(DATA)));
201 CHECK(isCleared(r.s));
202 }
203 { /* Bare error SW (unauthenticated, may be injected): cleared, so the next
204 * command cannot go out with a stale IV. */
205 Rig r;
206 r.nfc.sendOk = true;
207 r.nfc.resp[0] = 0x6FU;
208 r.nfc.resp[1] = 0x00U;
209 r.nfc.respLen = 2U;
210 CHECK(!r.send(DATA, sizeof(DATA)));
211 CHECK(isCleared(r.s));
212 }
213 { /* Response too short to hold an SW: cleared. */
214 Rig r;
215 r.nfc.sendOk = true;
216 r.nfc.respLen = 1U;
217 CHECK(!r.send(DATA, sizeof(DATA)));
218 CHECK(isCleared(r.s));
219 }
220 { /* Oversized payload, rejected before anything is sent: session kept. */
221 static const uint8_t big[CW_USER_DATA_PAGE_SIZE + 1U] = { 0U };
222 Rig r;
223 CHECK(!r.send(big, sizeof(big)));
224 CHECK(!isCleared(r.s) && (r.s.macCounter == 0U) && (r.nfc.sent == 0U));
225 }
226}

232static const uint8_t PATH[16] = {
233 0x80U, 0x00U, 0x00U, 0x2CU, 0x80U, 0x00U, 0x01U, 0xF5U,
234 0x80U, 0x00U, 0x00U, 0x00U, 0x80U, 0x00U, 0x00U, 0x00U

237/* connect() fails after SELECT (the fake answers nothing else), which leaves the
238 * wallet's applet version set; the session is then opened by hand. */
239static void selectOnly(CryptnoxWallet& wallet, FakeNfc& nfc, CW_SecureSession& s) {
240 const bool connected = wallet.connect(s);
241 CHECK(!connected);
242 nfc.sent = 0U;
243}

245static bool pathFor(uint8_t keyType) {
246 return ((keyType & 0x0FU) == 1U) || ((keyType & 0x0FU) == 2U);
247}
248
249static void check_pubkey_allow_list(void) {
250 FakeNfc nfc;
251 FakeCrypto crypto;
252 CryptnoxWallet wallet(nfc, logger, crypto, platform);
253 CW_SecureSession s;
254 selectOnly(wallet, nfc, s);
255
256 uint8_t pk[65] = { 0U };
257 uint16_t pkLen = 0U;
258
259 /* Refused locally: derive-and-make-current, PIN-less, unknown curve. */
260 const uint8_t refused[] = { 0x02U, 0x12U, 0x22U, 0x03U, 0x13U, 0x30U, 0x31U, 0xF0U };
261 for (uint8_t i = 0U; i < sizeof(refused); i++) {
262 const uint8_t t = refused[i];
263 openSession(s);
264 CHECK(!wallet.getPublicKey(s, t, pathFor(t) ? PATH : NULL, pathFor(t) ? sizeof(PATH) : 0U,
265 pk, sizeof(pk), pkLen));
266 CHECK((nfc.sent == 0U) && !isCleared(s));
267 }
268
269 /* Allowed: reaches the card with P1 = keyType (the fake then fails the exchange). */
270 const uint8_t allowed[] = { CW_SIGN_CURR_K1, CW_SIGN_CURR_R1, CW_SIGN_DERIVE_K1,
271 CW_SIGN_DERIVE_R1, CW_SIGN_CURR_ED25519, CW_SIGN_DERIVE_ED25519 };
272 for (uint8_t i = 0U; i < sizeof(allowed); i++) {
273 const uint8_t t = allowed[i];
274 openSession(s);
275 nfc.sent = 0U;
276 (void)wallet.getPublicKey(s, t, pathFor(t) ? PATH : NULL, pathFor(t) ? sizeof(PATH) : 0U,
277 pk, sizeof(pk), pkLen);
278 CHECK((nfc.sent == 1U) && (nfc.last[1] == 0xC2U) && (nfc.last[2] == t));
279 }
280}

286static CW_SignResult signWith(CryptnoxWallet& wallet, CW_SecureSession& s, uint8_t keyType) {
287 static const uint8_t msg[32] = { 0x01U };
288 CW_SignRequest req(s, keyType, CW_SIGN_SIG_ECDSA_LOW_S, CW_SIGN_WITH_PIN);
289 req.hash = msg;
290 req.hashLength = sizeof(msg);
291 memcpy(req.pin, "000000000", 9U);
292 return wallet.sign(req);
293}

295static void check_ed25519_applet_gate(void) {
296 /* Applet 1.6.x: refused before anything is sent. */
297 {
298 uint8_t pk[65] = { 0U };
299 uint16_t pkLen = 0U;
300 FakeNfc nfc;
301 nfc.appletMajor = 1U;
302 FakeCrypto crypto;
303 CryptnoxWallet wallet(nfc, logger, crypto, platform);
304 CW_SecureSession s;
305 selectOnly(wallet, nfc, s);
306
307 openSession(s);
308 CHECK(!wallet.getPublicKey(s, CW_SIGN_CURR_ED25519, NULL, 0U, pk, sizeof(pk), pkLen));
309 CHECK(nfc.sent == 0U);
310
311 openSession(s);
312 CHECK(signWith(wallet, s, CW_SIGN_CURR_ED25519).errorCode == CW_SIGN_UNSUPPORTED_KEY_TYPE);
313 CHECK(nfc.sent == 0U);
314
315 /* k1 is not affected. */
316 openSession(s);
317 (void)signWith(wallet, s, CW_SIGN_CURR_K1);
318 CHECK(nfc.sent == 1U);
319 }
320
321 /* Applet 2.x: sent, with P2 forced to EdDSA whatever signatureType says. */
322 {
323 FakeNfc nfc;
324 FakeCrypto crypto;
325 CryptnoxWallet wallet(nfc, logger, crypto, platform);
326 CW_SecureSession s;
327 selectOnly(wallet, nfc, s);
328
329 openSession(s);
330 CHECK(signWith(wallet, s, CW_SIGN_CURR_ED25519).errorCode != CW_SIGN_UNSUPPORTED_KEY_TYPE);
331 CHECK((nfc.sent == 1U) && (nfc.last[1] == 0xC0U) &&
332 (nfc.last[2] == CW_SIGN_CURR_ED25519) && (nfc.last[3] == CW_SIGN_SIG_EDDSA));
333 }
334}
