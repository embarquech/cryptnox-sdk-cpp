22/* True for the Ed25519 key types (high nibble 0x2 = ed25519 curve). */
23static bool cwIsEd25519(uint8_t keyType) {
24 return (keyType & 0xF0U) == 0x20U;
25}
26
27/* True when the SIGN key type carries a BIP32 derivation path
28 * (low nibble 1 = derive, 2 = derive-and-make-current). */
29static bool cwSignNeedsPath(uint8_t keyType) {
30 uint8_t derivation = keyType & 0x0FU;
31 return (derivation == 1U) || (derivation == 2U);
32}
33
34/* GET PUBLIC KEY is a read: only current / derive on k1, r1, ed25519.
35 * Excluded:
36 * - derive-and-make-current (low nibble 2): would change the card's current key;
37 * - PIN-less path (low nibble 3, CW_SIGN_PINLESS_K1): a SIGN-only mode, not a key to read. */
38static bool cwPubKeyTypeAllowed(uint8_t keyType) {
39 return (keyType == CW_SIGN_CURR_K1) || (keyType == CW_SIGN_DERIVE_K1) ||
40 (keyType == CW_SIGN_CURR_R1) || (keyType == CW_SIGN_DERIVE_R1) ||
41 (keyType == CW_SIGN_CURR_ED25519) || (keyType == CW_SIGN_DERIVE_ED25519);
42}
43
44/* Largest Ed25519 message that still fits one secure-channel page for THIS
45 * request. buildSignPayload() lays the data field out as
46 * [len(2)|msg|path?|pin(9)], so the room left for the message depends on the
47 * path the request actually carries.
48 *
49 * CW_MAX_ED25519_MESSAGE_LENGTH assumes the worst case,
50 * CW_MAX_DERIVE_PATH_LENGTH (20), and so rejects messages that do fit: a 4-level
51 * Solana path is 16 bytes, which leaves room for 181 rather than 177. That
52 * difference is what decides whether a ~180-byte SPL transfer can be signed with
53 * the path inside the SIGN command (P1 0x21) instead of requiring the key to be
54 * made current beforehand.
55 *
56 * The card itself is not the limit here: the v2.0 SIGN spec allows up to 1200
57 * bytes of raw data for P1 0x20/0x21. The 208-byte ceiling is this SDK's
58 * single-page secure-channel buffer. */
59static uint16_t cwMaxEd25519Message(const CW_SignRequest& request) {
60 /* Callers must have rejected an over-long path first (validateSignRequest
61 * does), otherwise the subtraction below would wrap. */
62 uint16_t overhead = 2U; /* big-endian message length prefix */
63
64 if (cwSignNeedsPath(request.keyType) && (request.derivePath != NULL)) {
65 overhead = (uint16_t)(overhead + request.derivePathLength);
66 }
67 if (!request.pinLessMode) {
68 overhead = (uint16_t)(overhead + CW_MAX_PIN_LENGTH);
69 }
70
71 return (uint16_t)(CW_USER_DATA_PAGE_SIZE - overhead);
72}

96bool CryptnoxWallet::connect(CW_SecureSession& session) {
97 bool ret = false;
98 session.clear(); /* CRIT-04: clear any stale keys from a previous or partial session */
99
100 for (uint8_t attempt = 0U; (attempt < CW_CONNECT_MAX_ATTEMPTS) && (ret == false); attempt++) {
101 if (attempt > 0U) {
102 session.clear(); /* CRIT-04: clear partial keys left by a failed attempt before retrying */
103#if CW_DEBUG_LOGGING
104 _logger.print(F("Retrying card connection (attempt "));
105 _logger.print((uint8_t)(attempt + 1U));
106 _logger.println(F(")..."));
107#endif
108 _secure.resetReader();
109 _platform.sleep_ms(200U);
110 }
111
112 if (_secure.inListPassiveTarget()) {
113 _platform.sleep_ms(200U);
114 if (establishSecureChannel(session)) {
115 ret = true;
116 }
117 }
118 }
119
120 if (!ret) {
121 session.clear(); /* CRIT-04: clear any partial keys from the final failed attempt */
122 }
123
124 return ret;
125}

127bool CryptnoxWallet::establishSecureChannel(CW_SecureSession& session) {
128 bool ret = false;
129
130 /* Declare all sensitive stack buffers at function entry so they can be
131 * wiped on every exit path (H-01, M-02). */
132 uint8_t cardCertificate[146U] = { 0U };
133 uint8_t cardCertificateLength = 0U;
134 uint8_t cardEphemeralPubKey[64U] = { 0U };
135 uint8_t openSecureChannelSalt[32U] = { 0U };
136 uint8_t clientPrivateKey[32U] = { 0U };
137 uint8_t clientPublicKey[64U] = { 0U };
138 CW_Curve sessionCurve = CW_CURVE_SECP256R1;
139
140 if (_secure.selectApdu()) {
141 /* Fetch the manufacturer certificate BEFORE getCardCertificate().
142 * The Cryptnox card state machine advances after GET_CARD_CERTIFICATE
143 * (INS=F8) and will not respond to GET_MANUFACTURER_CERTIFICATE (INS=F7)
144 * after that point. Pre-fetching here caches the cert inside
145 * CW_SecureChannel so that verifyCertificateChain() can use it without
146 * issuing another APDU. */
147 if (!_secure.preFetchManufacturerCert()) {
148#if CW_DEBUG_LOGGING
149 _logger.println(F("Failed to pre-fetch manufacturer certificate"));
150#endif
151 } else {
152 if (_secure.getCardCertificate(cardCertificate, cardCertificateLength)) {
153#if CW_VERIFY_CERT
154 uint8_t certResult = _secure.verifyCertificateChain(cardCertificate,
155 cardCertificateLength);
156#else
157 /* CW_VERIFY_CERT=0: the chain is not checked at all, so any card
158 * is accepted and will sign whatever it is given. Logged outside
159 * CW_DEBUG_LOGGING so the build can never do this silently. */
160 uint8_t certResult = CW_CERT_OK;
161 _logger.println(F("*** CW_VERIFY_CERT=0: card authenticity NOT verified."));
162#endif
163 if (certResult != CW_CERT_OK) {
164#if CW_DEBUG_LOGGING
165 _logger.print(F("Card authenticity check failed (code 0x"));
166 _logger.print(certResult, HEX);
167 _logger.println(F("). Aborting."));
168#endif
169 } else {
170 if (_secure.extractCardEphemeralKey(cardCertificate, cardEphemeralPubKey)) {
171 if (_secure.openSecureChannel(openSecureChannelSalt, clientPublicKey,
172 clientPrivateKey, sessionCurve)) {
173 if (_secure.mutuallyAuthenticate(session, openSecureChannelSalt,
174 clientPublicKey, clientPrivateKey,
175 sessionCurve, cardEphemeralPubKey)) {
176#if CW_DEBUG_LOGGING
177 _logger.println(F("Secure channel established"));
178#endif
179 ret = true;
180 } else {
181#if CW_DEBUG_LOGGING
182 _logger.println(F("Mutual authentication failed"));
183#endif
184 }
185 } else {
186#if CW_DEBUG_LOGGING
187 _logger.println(F("Failed to open secure channel"));
188#endif
189 }
190 } else {
191#if CW_DEBUG_LOGGING
192 _logger.println(F("Failed to extract card ephemeral key"));
193#endif
194 }
195 }
196 } else {
197#if CW_DEBUG_LOGGING
198 _logger.println(F("Failed to get card certificate"));
199#endif
200 }
201 } /* end preFetchManufacturerCert else */
202 } else {
203#if CW_DEBUG_LOGGING
204 _logger.println(F("Failed to select Cryptnox application"));
205#endif
206 }
207
208 /* Wipe all sensitive ephemeral key material on every exit path (H-01, M-02). */
209 CW_Utils::secure_wipe(clientPrivateKey, sizeof(clientPrivateKey));
210 CW_Utils::secure_wipe(openSecureChannelSalt, sizeof(openSecureChannelSalt));
211 CW_Utils::secure_wipe(clientPublicKey, sizeof(clientPublicKey));
212 CW_Utils::secure_wipe(cardEphemeralPubKey, sizeof(cardEphemeralPubKey));
213 CW_Utils::secure_wipe(cardCertificate, sizeof(cardCertificate));
214
215 return ret;
216}

225bool CryptnoxWallet::getCardInfo(CW_SecureSession& session, CW_CardInfo* info) {
226 bool ret = false;
227 if (!isSecureChannelOpen(session)) {
228#if CW_DEBUG_LOGGING
229 _logger.println(F("Error: Secure channel not open. Cannot get card info."));
230#endif
231 return false;
232 }
233 uint8_t data[] = { 0x00U };
234 uint8_t apdu[] = { 0x80U, 0xFAU, 0x00U, 0x00U };
235
236 uint8_t decrypted[255U] = { 0U };
237 uint16_t decryptedLen = 0U;
238
239 ret = _secure.aesCbcEncrypt(session, apdu, sizeof(apdu),
240 data, sizeof(data),
241 decrypted, sizeof(decrypted), &decryptedLen);
242
243 if (ret && (info != NULL)) {
244 /* Response layout (Cryptnox basic_g1 spec):
245 * [byte0] [name_len(1)] [name(name_len)]
246 * [email_len(1)] [email(email_len)] [... more fields ...]
247 * byte0 = unused/flags. */
248 ret = false;
249 if (decryptedLen >= 4U) {
250 uint16_t pos = 1U;
251 uint8_t nameLen = decrypted[pos];
252 pos += 1U;
253 if ((nameLen <= CW_CARD_NAME_MAX_LEN) &&
254 ((uint16_t)(pos + nameLen + 1U) <= decryptedLen)) {
255 (void)CW_Utils::safe_memcpy(reinterpret_cast<uint8_t*>(info->name),
256 sizeof(info->name),
257 decrypted + pos, nameLen);
258 info->name[nameLen] = '\0';
259 pos += nameLen;
260
261 uint8_t emailLen = decrypted[pos];
262 pos += 1U;
263 if ((emailLen <= CW_CARD_EMAIL_MAX_LEN) &&
264 ((uint16_t)(pos + emailLen) <= decryptedLen)) {
265 (void)CW_Utils::safe_memcpy(reinterpret_cast<uint8_t*>(info->email),
266 sizeof(info->email),
267 decrypted + pos, emailLen);
268 info->email[emailLen] = '\0';
269 ret = true;
270 }
271 }
272 }
273 }
274
275 CW_Utils::secure_wipe(decrypted, sizeof(decrypted));
276 return ret;
277}

279bool CryptnoxWallet::verifyPin(CW_SecureSession& session, const uint8_t* pin, uint8_t pinLength) {
280 bool ret = false;
281 if (!isSecureChannelOpen(session)) {
282#if CW_DEBUG_LOGGING
283 _logger.println(F("Error: Secure channel not open. Cannot verify PIN."));
284#endif
285 }
286 else if ((pin == NULL) || (pinLength < CW_MIN_PIN_LENGTH) || (pinLength > CW_MAX_PIN_LENGTH)) {
287#if CW_DEBUG_LOGGING
288 _logger.println(F("Error: Invalid PIN (must be 4-9 digits)."));
289#endif
290 }
291 else {
292 uint8_t paddedPin[CW_MAX_PIN_LENGTH] = { 0U };
293 (void)CW_Utils::safe_memcpy(paddedPin, sizeof(paddedPin), pin, pinLength);
294 uint8_t apdu[] = { 0x80U, 0x20U, 0x00U, 0x00U };
295 ret = _secure.aesCbcEncrypt(session, apdu, sizeof(apdu), paddedPin, CW_MAX_PIN_LENGTH);
296 CW_Utils::secure_wipe(paddedPin, sizeof(paddedPin));
297 }
298 return ret;

302bool CryptnoxWallet::writeUserData(CW_SecureSession& session,
303 const uint8_t* data, uint16_t dataLength) {
304 bool ret = false;
305
306 if (!isSecureChannelOpen(session)) {
307#if CW_DEBUG_LOGGING
308 _logger.println(F("Error: Secure channel not open. Cannot write user data."));
309#endif
310 }
311 else if ((data == NULL) || (dataLength == 0U)) {
312#if CW_DEBUG_LOGGING
313 _logger.println(F("Error: Invalid data for write user data."));
314#endif
315 }
316 else if (dataLength > CW_USER_DATA_PAGE_SIZE) {
317 /* The applet's WRITE (0xFC) P2 selects a 1200-byte logical page written
318 * at offset 1200*P2, not a chunk counter — so multiple single-APDU
319 * chunks cannot be concatenated over this transport. Reject anything
320 * that won't fit one page instead of silently corrupting the write. */
321#if CW_DEBUG_LOGGING
322 _logger.print(F("Error: user data exceeds one page ("));
323 _logger.print((uint8_t)CW_USER_DATA_PAGE_SIZE);
324 _logger.println(F(" B max for this transport)."));
325#endif
326 }
327 else {
328 /* P1 = 0 (write user data slot; P1 is the applet's operation selector,
329 * not a slot index), P2 = 0 (logical page 0). */
330 uint8_t apdu[] = { 0x80U, 0xFCU, 0x00U, 0x00U };
331 ret = _secure.aesCbcEncrypt(session, apdu, sizeof(apdu), data, dataLength);
332#if CW_DEBUG_LOGGING
333 if (!ret) {
334 _logger.println(F("Error: Write user data failed."));
335 }
336#endif
337 }
338
339 return ret;
340}

342bool CryptnoxWallet::getPublicKey(CW_SecureSession& session,
343 const uint8_t* derivePath,
344 uint8_t derivePathLength,
345 uint8_t* publicKey64) {
346 bool ret = false;
347 /* SEC1 uncompressed point: 0x04 || X || Y. Some applet versions answer
348 * with the bare coordinates, so accept both. */
349 uint8_t key[CW_SEC1_PUBKEY_SIZE] = { 0U };
350 uint16_t keyLength = 0U;
351 const uint8_t keyType = (derivePath != NULL) ? CW_SIGN_DERIVE_K1 : CW_SIGN_CURR_K1;
352
353 if ((publicKey64 == NULL) ||
354 ((derivePath == NULL) != (derivePathLength == 0U))) {
355#if CW_DEBUG_LOGGING
356 _logger.println(F("Error: Invalid arguments for get public key."));
357#endif
358 }
359 else if (getPublicKey(session, keyType, derivePath, derivePathLength,
360 key, sizeof(key), keyLength)) {
361 if ((keyLength == CW_SEC1_PUBKEY_SIZE) && (key[0] == CW_SEC1_UNCOMPRESSED_TAG)) {
362 ret = CW_Utils::safe_memcpy(publicKey64, CW_RAW_PUBKEY_SIZE, &key[1], CW_RAW_PUBKEY_SIZE);
363 }
364 else if (keyLength == CW_RAW_PUBKEY_SIZE) {
365 ret = CW_Utils::safe_memcpy(publicKey64, CW_RAW_PUBKEY_SIZE, key, CW_RAW_PUBKEY_SIZE);
366 }
367 else {
368#if CW_DEBUG_LOGGING
369 _logger.println(F("Get public key: unexpected response length."));
370#endif
371 }
372 }
373 else {
374 /* getPublicKey() above already logged the cause. */
375 }
376
377 CW_Utils::secure_wipe(key, sizeof(key));
378 return ret;
379}

381CW_SignResult CryptnoxWallet::sign(CW_SignRequest& request) {
382 CW_SignResult result{};
383
384 if (validateSignRequest(request, result)) {
385 /* One secure-channel page holds the largest framed payload for either
386 * scheme: ECDSA [hash|path|pin] or Ed25519 [len(2)|msg|path|pin]. */
387 uint8_t data[CW_USER_DATA_PAGE_SIZE] = { 0U };
388 uint16_t dataLength = 0U;
389
390 buildSignPayload(request, data, dataLength);
391
392 uint8_t derResponse[255U] = { 0U };
393 uint16_t derLength = 0U;
394
395 if (sendSignApdu(request, data, dataLength, derResponse, sizeof(derResponse),
396 derLength, result)) {
397 /* Ed25519 (EdDSA) and Schnorr BIP340 both return a raw 64-byte
398 * R||S signature with no DER wrapper; ECDSA (low-S / EOSIO) returns
399 * a DER SEQUENCE that extractRawSignature() unpacks. */
400 const bool rawSig = cwIsEd25519(request.keyType) ||
401 (request.signatureType == CW_SIGN_SIG_SCHNORR_BIP340);
402 if (rawSig) {
403 if (derLength == CW_RAW_SIGNATURE_SIZE) {
404 (void)CW_Utils::safe_memcpy(result.signature, sizeof(result.signature),
405 derResponse, CW_RAW_SIGNATURE_SIZE);
406 debugPrintSignature(result.signature);
407 result.errorCode = CW_OK;
408 } else {
409#if CW_DEBUG_LOGGING
410 _logger.println(F("Error: raw signature not 64 bytes."));
411#endif
412 result.errorCode = CW_NOK;
413 }
414 } else if (extractRawSignature(derResponse, derLength, result)) {
415 debugPrintSignature(result.signature);
416 result.errorCode = CW_OK;
417 }
418 }
419 CW_Utils::secure_wipe(data, sizeof(data));
420 CW_Utils::secure_wipe(derResponse, sizeof(derResponse));
421 }
422
423 return result;
424}

426bool CryptnoxWallet::getPublicKey(CW_SecureSession& session, uint8_t keyType,
427 const uint8_t* derivePath, uint8_t derivePathLength,
428 uint8_t* pubKey, uint16_t pubKeyBufSize,
429 uint16_t& pubKeyLength) {
430 bool ret = false;
431 pubKeyLength = 0U;
432
433 if (!isSecureChannelOpen(session)) {
434#if CW_DEBUG_LOGGING
435 _logger.println(F("Error: Secure channel not open. Cannot get public key."));
436#endif
437 }
438 else if ((pubKey == NULL) || (pubKeyBufSize == 0U)) {
439#if CW_DEBUG_LOGGING
440 _logger.println(F("Error: Invalid output buffer for get public key."));
441#endif
442 }
443 else if (!cwPubKeyTypeAllowed(keyType)) {
444#if CW_DEBUG_LOGGING
445 _logger.println(F("Error: Unsupported key type for get public key."));
446#endif
447 }
448 else if (cwIsEd25519(keyType) && (_secure.appletMajor() < 2U)) {
449#if CW_DEBUG_LOGGING
450 _logger.println(F("Error: Ed25519 requires applet 2.0+."));
451#endif
452 }
453 else if (cwSignNeedsPath(keyType) &&
454 ((derivePath == NULL) || (derivePathLength == 0U) ||
455 (derivePathLength > CW_MAX_DERIVE_PATH_LENGTH) ||
456 ((derivePathLength % 4U) != 0U))) {
457#if CW_DEBUG_LOGGING
458 _logger.println(F("Error: Invalid derivation path for get public key."));
459#endif
460 }
461 else {
462 /* GET PUBLIC KEY (APDU 0x80C2): P1 = derivation|curve, P2 = 0x01.
463 * Data is the BIP32 path for DERIVE key types, empty for current-key. */
464 uint8_t apdu[] = { 0x80U, 0xC2U, keyType, 0x01U };
465 const uint8_t* data = cwSignNeedsPath(keyType) ? derivePath : NULL;
466 uint16_t dataLen = cwSignNeedsPath(keyType) ? derivePathLength : 0U;
467
468 uint8_t decrypted[255U] = { 0U };
469 uint16_t decryptedLen = 0U;
470
471 if (_secure.aesCbcEncrypt(session, apdu, sizeof(apdu), data, dataLen,
472 decrypted, sizeof(decrypted), &decryptedLen)) {
473 if ((decryptedLen > 0U) && (decryptedLen <= pubKeyBufSize)) {
474 (void)CW_Utils::safe_memcpy(pubKey, pubKeyBufSize, decrypted, decryptedLen);
475 pubKeyLength = decryptedLen;
476 ret = true;
477 }
478#if CW_DEBUG_LOGGING
479 else {
480 _logger.println(F("Error: public key buffer too small or empty response."));
481 }
482#endif
483 }
484 CW_Utils::secure_wipe(decrypted, sizeof(decrypted));
485 }
486
487 return ret;
488}

554bool CryptnoxWallet::validateSignRequest(const CW_SignRequest& request, CW_SignResult& result) {
555 bool ret = false;
556
557 if (!isSecureChannelOpen(request.session)) {
558#if CW_DEBUG_LOGGING
559 _logger.println(F("Error: Secure channel not open. Cannot sign."));
560#endif
561 result.errorCode = CW_INVALID_SESSION;
562 }
563 else if ((request.hash == NULL) || (request.hashLength == 0U)) {
564#if CW_DEBUG_LOGGING
565 _logger.println(F("Error: Invalid parameters for sign."));
566#endif
567 result.errorCode = CW_SIGN_KEY_TOO_SHORT;
568 }
569 else if (cwIsEd25519(request.keyType) && (_secure.appletMajor() < 2U)) {
570#if CW_DEBUG_LOGGING
571 _logger.println(F("Error: Ed25519 requires applet 2.0+."));
572#endif
573 result.errorCode = CW_SIGN_UNSUPPORTED_KEY_TYPE;
574 }
575 /* Validate the path BEFORE sizing the message, which subtracts its length
576 * from the page: an out-of-range path would make that subtraction wrap and
577 * report a nonsense cap. getPublicKey() already applies these same rules. */
578 else if (cwSignNeedsPath(request.keyType) &&
579 ((request.derivePath == NULL) || (request.derivePathLength == 0U) ||
580 (request.derivePathLength > CW_MAX_DERIVE_PATH_LENGTH) ||
581 ((request.derivePathLength % 4U) != 0U))) {
582#if CW_DEBUG_LOGGING
583 _logger.println(F("Error: Invalid derivation path for sign."));
584#endif
585 result.errorCode = CW_SIGN_INVALID_PATH;
586 }
587 else if (request.hashLength > (cwIsEd25519(request.keyType)
588 ? cwMaxEd25519Message(request)
589 : CW_HASH_SIZE)) {
590#if CW_DEBUG_LOGGING
591 _logger.println(F("Error: Message/hash too large to sign."));
592#endif
593 result.errorCode = CW_SIGN_MESSAGE_TOO_LONG;
594 }
595 else if ((request.pinLessMode) && (request.keyType != CW_SIGN_PINLESS_K1)) {
596#if CW_DEBUG_LOGGING
597 _logger.println(F("Error: PIN-less mode requires CW_SIGN_PINLESS_K1 key type."));
598#endif
599 result.errorCode = CW_SIGN_KEY_TOO_SHORT_WITH_PINLESS_MODE;
600 }
601 else {
602 ret = true;
603
604 if (!request.pinLessMode) {
605 uint8_t pinLength = 0U;
606 for (uint8_t i = 0U; i < CW_MAX_PIN_LENGTH; i++) {
607 if (request.pin[i] == 0U) { break; }
608 pinLength++;
609 }
610 if ((pinLength > 0U) && (pinLength < CW_MIN_PIN_LENGTH)) {
611#if CW_DEBUG_LOGGING
612 _logger.println(F("Error: PIN too short (must be 4-9 digits)."));
613#endif
614 result.errorCode = CW_SIGN_PIN_INCORRECT;
615 ret = false;
616 }
617 }
618 }
619
620 return ret;
621}

623void CryptnoxWallet::buildSignPayload(const CW_SignRequest& request,
624 uint8_t* data, uint16_t& dataLength) {
625 const size_t kDataBufSize = static_cast<size_t>(CW_USER_DATA_PAGE_SIZE);
626 dataLength = 0U;
627
628 if (cwIsEd25519(request.keyType)) {
629 /* EdDSA data field (docs.cryptnox.com v2.0 SIGN command):
630 * [2-byte big-endian message length][raw message][path?][PIN(9B, 0x00-padded)]
631 * The card hashes the raw message internally, so the length prefix is
632 * what delimits the variable-length message from the trailing fields. */
633 data[0] = (uint8_t)((request.hashLength >> 8) & 0xFFU);
634 data[1] = (uint8_t)(request.hashLength & 0xFFU);
635 dataLength = 2U;
636 }
637
638 (void)CW_Utils::safe_memcpy(data + dataLength, kDataBufSize - static_cast<size_t>(dataLength),
639 request.hash, request.hashLength);
640 dataLength += request.hashLength;
641
642 if (cwSignNeedsPath(request.keyType) &&
643 (request.derivePath != NULL) && (request.derivePathLength > 0U)) {
644 (void)CW_Utils::safe_memcpy(data + dataLength, kDataBufSize - static_cast<size_t>(dataLength),
645 request.derivePath, request.derivePathLength);
646 dataLength += request.derivePathLength;
647 }
648
649 if (!request.pinLessMode) {
650 uint8_t pinLength = 0U;
651 for (uint8_t i = 0U; i < CW_MAX_PIN_LENGTH; i++) {
652 if (request.pin[i] == 0U) { break; }
653 pinLength++;
654 }
655 if (pinLength > 0U) {
656 /* PIN is a fixed 9-byte field, right-padded with 0x00. */
657 (void)CW_Utils::safe_memcpy(data + dataLength, kDataBufSize - static_cast<size_t>(dataLength),
658 request.pin, CW_MAX_PIN_LENGTH);
659 dataLength += CW_MAX_PIN_LENGTH;
660 }
661 }
662}

664bool CryptnoxWallet::sendSignApdu(CW_SignRequest& request, const uint8_t* data,
665 uint16_t dataLength, uint8_t* derResponse,
666 uint16_t derResponseSize,
667 uint16_t& derLength, CW_SignResult& result) {
668 bool ret = false;
669 /* Ed25519 forces P2 = EdDSA; the caller-supplied signatureType is ignored. */
670 uint8_t p2 = cwIsEd25519(request.keyType) ? CW_SIGN_SIG_EDDSA : request.signatureType;
671 uint8_t apdu[] = { 0x80U, 0xC0U, request.keyType, p2 };
672
673#if CW_DEBUG_LOGGING
674 _logger.println(F("Sending SIGN APDU..."));
675#endif
676
677 if (_secure.aesCbcEncrypt(request.session, apdu, sizeof(apdu), data, dataLength,
678 derResponse, derResponseSize, &derLength)) {
679 ret = true;
680 }
681 else {
682#if CW_DEBUG_LOGGING
683 _logger.println(F("Sign APDU failed."));
684#endif
685 result.errorCode = CW_SIGN_NO_KEY_LOADED;
686 }
687
688 return ret;
689}

691bool CryptnoxWallet::extractRawSignature(const uint8_t* derResponse, uint16_t derLength,
692 CW_SignResult& result) {
693 bool ret = false;
694
695 if ((derLength < 2U) || (derResponse[0] != CW_DER_TAG_SEQUENCE)) {
696#if CW_DEBUG_LOGGING
697 _logger.println(F("Error: Invalid signature data (missing DER SEQUENCE tag)."));
698#endif
699 result.errorCode = CW_NOK;
700 }
701 else {
702 uint8_t derContentLength = derResponse[1];
703 uint8_t derTotalLength = 2U + derContentLength;
704
705 if (derTotalLength > derLength) {
706#if CW_DEBUG_LOGGING
707 _logger.println(F("Error: DER signature length exceeds response."));
708#endif
709 result.errorCode = CW_NOK;
710 }
711 else {
712 uint8_t r[33U] = { 0U };
713 uint8_t s[33U] = { 0U };
714 uint8_t rLen = 0U;
715 uint8_t sLen = 0U;
716
717 if (!parseDerSignature(derResponse, derTotalLength, r, rLen, s, sLen)) {
718#if CW_DEBUG_LOGGING
719 _logger.println(F("Error: Failed to parse DER signature."));
720#endif
721 result.errorCode = CW_NOK;
722 }
723 else {
724 memset(result.signature, 0U, CW_RAW_SIGNATURE_SIZE);
725
726 if (rLen > 0U) {
727 uint8_t rSrc = 0U;
728 uint8_t rDstLen = 32U;
729 if ((rLen == 33U) && (r[0] == 0x00U)) { rSrc = 1U; rLen = 32U; }
730 if (rLen <= rDstLen) {
731 (void)CW_Utils::safe_memcpy(result.signature + (rDstLen - rLen), static_cast<size_t>(rDstLen + rLen), r + rSrc, rLen);
732 }
733 }
734
735 if (sLen > 0U) {
736 uint8_t sSrc = 0U;
737 uint8_t sDstLen = 32U;
738 if ((sLen == 33U) && (s[0] == 0x00U)) { sSrc = 1U; sLen = 32U; }
739 if (sLen <= sDstLen) {
740 (void)CW_Utils::safe_memcpy(result.signature + 32U + (sDstLen - sLen), static_cast<size_t>(sLen), s + sSrc, sLen);
741 }
742 }
743
744 ret = true;
745 }
746 CW_Utils::secure_wipe(r, sizeof(r));
747 CW_Utils::secure_wipe(s, sizeof(s));
748 }
749 }
750
751 return ret;
752}
